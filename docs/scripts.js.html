/**
 * Clase que representa una tarea.
 * Cada tarea tiene un texto y un estado (completada o no).
 */
class Task {
    /**
     * Constructor de la clase Task.
     * @param {string} text - El texto de la tarea que se va a crear.
     */
    constructor(text) {
        this.text = text; // Guarda el texto de la tarea.
        this.completed = false; // Por defecto, la tarea empieza como "no completada".
    }
}

/**
 * Clase que maneja la lista de tareas y la sincronización con localStorage.
 */
class TaskManager {
    /**
     * Constructor de la clase TaskManager.
     * Recupera las tareas almacenadas en localStorage o inicializa una lista vacía.
     */
    constructor() {
        this.tasks = JSON.parse(localStorage.getItem('tasks')) || []; // Carga las tareas desde el almacenamiento local.
    }

    /**
     * Agrega una nueva tarea a la lista y la guarda en localStorage.
     * @param {string} text - El texto de la tarea nueva.
     */
    addTask(text) {
        const task = new Task(text); // Crea una nueva tarea.
        this.tasks.push(task); // La mete en el array de tareas.
        this.updateLocalStorage(); // Guarda la lista actualizada en localStorage.
    }

    /**
     * Elimina una tarea de la lista por su índice y actualiza localStorage.
     * @param {number} index - La posición de la tarea en el array.
     */
    removeTask(index) {
        this.tasks.splice(index, 1); // Borra la tarea en la posición indicada.
        this.updateLocalStorage(); // Guarda los cambios.
    }

    /**
     * Alterna el estado de completado de una tarea y actualiza localStorage.
     * @param {number} index - La posición de la tarea en el array.
     */
    toggleTaskCompleted(index) {
        this.tasks[index].completed = !this.tasks[index].completed; // Cambia entre "completado" y "no completado".
        this.updateLocalStorage(); // Guarda los cambios.
    }

    /**
     * Guarda la lista de tareas actualizada en localStorage.
     */
    updateLocalStorage() {
        localStorage.setItem('tasks', JSON.stringify(this.tasks)); // Convierte la lista a texto y la guarda.
    }

    /**
     * Devuelve la lista de tareas.
     * @returns {Array<Task>} Lista de tareas almacenadas.
     */
    getTasks() {
        return this.tasks; // Retorna la lista de tareas.
    }
}

// Se crea una instancia de TaskManager para manejar las tareas.
const taskManager = new TaskManager();

/**
 * Función que se ejecuta cuando se quiere agregar una nueva tarea desde la interfaz.
 */
function addTask() {
    const taskInput = document.getElementById('taskInput'); // Obtiene el input de tareas.
    const text = taskInput.value.trim(); // Obtiene el texto y elimina espacios en blanco innecesarios.
    if (text) { // Si el input no está vacío...
        taskManager.addTask(text); // Se añade la tarea con el texto ingresado.
        taskInput.value = ''; // Se limpia el input.
        renderTasks(); // Se vuelve a renderizar la lista de tareas.
    }
}

/**
 * Función para eliminar una tarea de la lista.
 * @param {number} index - La posición de la tarea en el array.
 */
function deleteTask(index) {
    taskManager.removeTask(index); // Se elimina la tarea del array.
    renderTasks(); // Se actualiza la lista en pantalla.
}

/**
 * Renderiza la lista de tareas en la interfaz.
 */
function renderTasks() {
    const taskList = document.getElementById('taskList'); // Obtiene la lista en el HTML.
    taskList.innerHTML = ''; // Limpia la lista antes de volver a generarla.

    // Se recorre cada tarea y se genera su HTML
    taskManager.getTasks().forEach((task, index) => {
        const taskEl = document.createElement('li'); // Se crea un elemento de lista.
        const taskText = document.createElement('span'); // Se crea un span para el texto de la tarea.
        taskText.textContent = task.text; // Se le asigna el texto de la tarea.
        taskText.style.flexGrow = '1'; // Se usa flex-grow para que ocupe todo el espacio disponible.

        // Si la tarea está marcada como completada, se le pone tachado.
        if (task.completed) {
            taskText.style.textDecoration = 'line-through';
        }
        
        // Se crea el botón para eliminar la tarea.
        const deleteBtn = document.createElement('button');
        deleteBtn.textContent = 'Borrar'; // Texto del botón.
        deleteBtn.onclick = () => deleteTask(index); // Se enlaza la función de borrar.
        deleteBtn.style.marginLeft = '10px'; // Margen para separarlo del texto.
        deleteBtn.classList.add('buttonB'); // Se le agrega una clase CSS.

        // Se añaden los elementos al item de la lista.
        taskEl.appendChild(taskText);
        taskEl.appendChild(deleteBtn);
        taskList.appendChild(taskEl); // Se agrega la tarea a la lista en el DOM.
    });
}

/**
 * Función para marcar o desmarcar una tarea como completada.
 * @param {number} index - La posición de la tarea en el array.
 */
function toggleTaskCompleted(index) {
    taskManager.toggleTaskCompleted(index); // Alterna el estado de la tarea.
    renderTasks(); // Se vuelve a renderizar la lista para reflejar el cambio.
}

// Se añade un evento al botón "Agregar" para llamar a la función addTask().
document.getElementById('addTaskBtn').addEventListener('click', addTask);

// Se renderizan las tareas guardadas al cargar la página.
renderTasks();
